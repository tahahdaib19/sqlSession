CREATE OR REPLACE PACKAGE DEPARTMENT_PKG AS

    TYPE NAMES_ARRAY IS TABLE OF DEPARTMENT.DEPT_NAME%TYPE
        INDEX BY PLS_INTEGER;

    PROCEDURE ADD_LIST_OF_DEPARTMENTS(
        P_NAMES IN NAMES_ARRAY
    );

END DEPARTMENT_PKG;
/


------------------------------- DEPARTMENT BODY
CREATE OR REPLACE PACKAGE BODY DEPARTMENT_PKG AS

    PROCEDURE ADD_LIST_OF_DEPARTMENTS(
        P_NAMES IN NAMES_ARRAY
    )
    IS
        V_INDEX PLS_INTEGER;
        V_COUNT PLS_INTEGER := 0;
    BEGIN
        V_INDEX := P_NAMES.FIRST;

        WHILE V_INDEX IS NOT NULL LOOP

            INSERT INTO DEPARTMENT (DEPT_NAME)
            VALUES (P_NAMES(V_INDEX));

            V_COUNT := V_COUNT + SQL%ROWCOUNT;

            V_INDEX := P_NAMES.NEXT(V_INDEX);

        END LOOP;

        DBMS_OUTPUT.PUT_LINE(
            'Departments inserted: ' || V_COUNT
        );

        COMMIT;

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;

            DBMS_OUTPUT.PUT_LINE(
                'Error occurred: ' || SQLERRM
            );

            RAISE;

    END ADD_LIST_OF_DEPARTMENTS;

END DEPARTMENT_PKG;

PROCEDURE UPDATE_EMPLOYEE (
    p_emp_id      EMPLOYEE.EMP_ID%TYPE,
    p_emp_name    EMPLOYEE.EMP_NAME%TYPE,
    p_emp_age     EMPLOYEE.EMP_AGE%TYPE,
    p_emp_dept_id EMPLOYEE.EMP_DEPT_ID%TYPE,
    p_salary      EMPLOYEE.SALARY%TYPE,
    p_manager_id  EMPLOYEE.MANAGER_ID%TYPE
)
IS
    v_dept_id EMPLOYEE.EMP_DEPT_ID%TYPE;
BEGIN
    SELECT d.DEPT_ID
    INTO v_dept_id
    FROM DEPARTMENT d
    WHERE d.DEPT_ID = p_emp_dept_id;

    UPDATE EMPLOYEE e
    SET e.EMP_NAME    = p_emp_name,
        e.EMP_AGE     = p_emp_age,
        e.EMP_DEPT_ID = v_dept_id,
        e.SALARY      = p_salary,
        e.MANAGER_ID  = p_manager_id
    WHERE e.EMP_ID = p_emp_id;

    DBMS_OUTPUT.PUT_LINE(
        'Rows affected = ' || SQL%ROWCOUNT
    );

    IF SQL%ROWCOUNT = 0 THEN
        DBMS_OUTPUT.PUT_LINE(
            'No employee found with ID = ' || p_emp_id
        );
    END IF;

    COMMIT;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE(
            'Department not found: ' || p_emp_dept_id
        );
        ROLLBACK;
END UPDATE_EMPLOYEE;
