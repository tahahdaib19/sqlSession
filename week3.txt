DECLARE
    TYPE EMP_IDS_TYPE IS TABLE OF NUMBER
        INDEX BY PLS_INTEGER;

    V_IDS EMP_IDS_TYPE;
BEGIN
    V_IDS(1) := 1;
    V_IDS(2) := 2;
    V_IDS(3) := 5;

    FOR I IN 1 .. V_IDS.COUNT LOOP
        UPDATE EMPLOYEE
        SET SALARY = SALARY + 100
        WHERE EMP_ID = V_IDS(I);
        END LOOP;

    DBMS_OUTPUT.PUT_LINE(
        'Total rows affected: ' || SQL%ROWCOUNT
    );

    COMMIT;
END;
