
CREATE OR REPLACE PROCEDURE NUM_PLUS_FIVE (
    v_input IN NUMBER,
    v_result OUT NUMBeR
)
AS
BEGIN
   v_result := v_input +5;
END;


DECLARE 
v_result NUMBER;
v_input number;
BEGIN 
v_result:=0;
v_input :=10;
NUM_PLUS_FIVE(v_input,v_result);
DBMS_OUTPUT.PUT_LINE( 'input'|| v_input);

DBMS_OUTPUT.PUT_LINE( 'out'|| v_result);
end ;

  CREATE OR REPLACE PACKAGE BODY EMPLOYEE_PKG
  AS

  --INSERT
 PROCEDURE ADD_EMPLOYEE (emp_name EMPLOYEE.EMP_NAME%TYPE ,emp_age EMPLOYEE.EMP_AGE%TYPE,EMP_dept_id EMPLOYEE.EMP_DEPT_ID%TYPE
  ,salary EMPLOYEE.SALARY%TYPE,manager_id EMPLOYEE.MANAGER_ID%TYPE)
 IS
 
 EMP_D_ID EMPLOYEE.EMP_DEPT_ID%TYPE;
 EMP_M_ID EMPLOYEE.MANAGER_ID%TYPE;
 BEGIN
    SELECT dept_id INTO EMP_D_ID  FROM DEPARTMENT WHERE dept_id=EMP_dept_id;
    SELECT EMP_ID INTO EMP_M_ID FROM EMPLOYEE WHERE  EMP_ID = manager_id;
     
     INSERT INTO EMPLOYEE (
            EMP_NAME,
            EMP_AGE,
            EMP_DEPT_ID,
            SALARY,
            MANAGER_ID
        )
        VALUES (
            emp_name,
            emp_age,
            EMP_dept_id,
            salary,
            manager_id
        );

        DBMS_OUTPUT.PUT_LINE('Employee added successfully.');
        COMMIT;

    EXCEPTION
        WHEN no_data_found THEN
            DBMS_OUTPUT.PUT_LINE('Error: manager id or dept id is not valid' );
            ROLLBACK;
    END ADD_EMPLOYEE;

 PROCEDURE UPDATE_EMPLOYEE
  (emp_id EMPLOYEE.EMP_ID %TYPE,emp_name EMPLOYEE.EMP_NAME%TYPE ,emp_age EMPLOYEE.EMP_AGE%TYPE,emp_dept_id EMPLOYEE.EMP_DEPT_ID%TYPE
  ,salary EMPLOYEE.SALARY%TYPE,manager_id EMPLOYEE.MANAGER_ID%TYPE)
    
    IS

 EMP_D_ID EMPLOYEE.EMP_DEPT_ID%TYPE;
 EMP_M_ID EMPLOYEE.MANAGER_ID%TYPE;
   

  BEGIN


    SELECT dept_id INTO EMP_D_ID  FROM DEPARTMENT WHERE dept_id=emp_dept_id;
    --SELECT EMP_ID INTO EMP_M_ID FROM EMPLOYEE WHERE  EMP_ID = manager_id;

    UPDATE EMPLOYEE 
    SET 
    EMP_NAME = emp_name,

            ROLLBACK;
  END UPDATE_EMPLOYEE;

  PROCEDURE DELETE_EMPLOYEE (ID NUMBER)
  IS
    BEGIN
        DELETE FROM EMPLOYEE WHERE EMP_ID = ID;
     IF SQL%ROWCOUNT = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Employee not found.');
        ELSE
            DBMS_OUTPUT.PUT_LINE('Employee deleted successfully.');
            COMMIT;
        END IF;

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
    END DELETE_EMPLOYEE;

    PROCEDURE GET_EMPLOYEES
    IS
    CURSOR C_EMPS IS SELECT *FROM EMPLOYEE   FETCH FIRST 1 ROW ONLY;    
    BEGIN
            FOR emp_record IN C_EMPS LOOP

        DBMS_OUTPUT.PUT_LINE(
            'ID: ' || emp_record.EMP_ID ||
            ' | Name: ' || emp_record.EMP_NAME ||
            ' | Age: ' || emp_record.EMP_AGE ||
            ' | Department: ' || emp_record.EMP_DEPT_ID ||
            ' | Salary: ' || emp_record.SALARY ||
            ' | Manager: ' || emp_record.MANAGER_ID
        );
    END LOOP;

    END GET_EMPLOYEES;
